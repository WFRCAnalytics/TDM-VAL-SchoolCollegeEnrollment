Overview

This script builds the geographic crosswalk that every later step in this project joins against.

  • Objective: tie every Census block in the Wasatch Front region (Box Elder, Davis, Weber, Salt Lake, and Utah counties) to the school district and county it falls in, and to the K-12 school(s) and college(s) located in it — so that PopulationSim’s synthetic population (3_synthetic.qmd) and real enrollment data (2_enrollment.qmd) can both be rolled up to district/county level as exact joins, with no area-weighting or estimation involved.
  • What it pulls: county, school district, Travel Analysis Zone (TAZ), and Census block boundaries; K-12 school and college locations from both UGRC/SGID (the state-preferred source) and NCES (kept alongside, not as a replacement, so schools and colleges can also be joined back to federal CCD/IPEDS data by ID); and the WFRC/MAG travel demand model boundary, used later to handle Box Elder County’s partial coverage in that model. TAZ boundaries come from WFRC’s WasatchFrontTAZ feature service rather than the TDM-GEO-AnalysisZones repo directly, so the pull stays current without tracking that repo’s releases — TAZID is what is kept as a reference layer. The synthetic join uses MAZ, not TAZ. MAZ boundaries come from a draft microzone layer supplied for Utah (_data/raw/geometry/microzones_draft_*.zip), not yet a published WFRC layer. Block boundaries are still cached and checked, but no crosswalk uses them.
  • Output: two crosswalks, one per use. _data/intermediate/crosswalk_school.csv maps each K-12 school (UGRC, NCES public, NCES private) and each MAZ to its district. _data/intermediate/crosswalk_college.csv maps each college (UGRC, NCES) and each MAZ to its county. MAZ rows appear in both files, so PopulationSim output rolls up through MAZ rather than through blocks. Section 10 checks coverage and maps the study area (Figure 1).

taz_boundary_path = paths["raw"] / "geometry" / "taz_boundary.gpkg"

# 1. Download only if not already cached locally
if not taz_boundary_path.exists():
    taz_url = (
        "https://services1.arcgis.com/taguadKoI1XFwivx/ArcGIS/rest/services/"
        "WasatchFrontTAZ/FeatureServer/0/query"
    )
    taz_pages = []
    offset = 0
    while True:
        page = gpd.read_file(
            f"{taz_url}?where=1=1&outFields=*&f=geojson"
            f"&orderByFields=FID&resultOffset={offset}&resultRecordCount=2000"
        )
        if page.empty:
            break
        taz_pages.append(page)
        offset += len(page)
        if len(page) < 2000:
            break
    all_taz = pd.concat(taz_pages, ignore_index=True).to_crs(constants["crs"])

    count_resp = requests.get(f"{taz_url}?where=1=1&returnCountOnly=true&f=json").json()
    assert len(all_taz) == count_resp["count"], (len(all_taz), count_resp["count"])

    all_taz["county_fips"] = all_taz["CO_FIPS"].astype(int).astype(str).str.zfill(3)

    all_taz.to_file(taz_boundary_path)

# 2. Read + filter in one step, pushed down to SQLite

5. Census blocks

Show code 
block_boundary_path = paths["raw"] / "geometry" / "block_boundary.gpkg"
block_counties_path = paths["raw"] / "geometry" / "block_boundary.counties.txt"

# Auto-invalidate the cache if constants["counties"] has changed since it was built -- pygris'
# blocks() call below is scoped to the CURRENT county list at fetch time (unlike every other
# cached layer in this file, which fetches statewide and filters at read-time), so a stale cache
# here would silently miss any newly-added county's blocks instead of erroring.
current_counties = sorted(county_fips)
cached_counties = (
    block_counties_path.read_text().splitlines() if block_counties_path.exists() else []
)
if block_boundary_path.exists() and current_counties != cached_counties:
    block_boundary_path.unlink()

# 1. Download only if not already cached locally
if not block_boundary_path.exists():
    all_blocks = blocks(state=state_fips, county=county_fips, year=2023, cache=True).to_crs(
        constants["crs"]
    )

    all_blocks["district_name"] = gpd.sjoin(
        all_blocks.set_geometry(all_blocks.centroid),
        bound_districts[["name", "geometry"]],
        predicate="within",
        how="left",
    )["name"]

    all_blocks.to_file(block_boundary_path)
    block_counties_path.write_text("\n".join(current_counties))

# 2. Read + filter in one step, pushed down to SQLite
county_list_sql = ", ".join(f"'{fips}'" for fips in county_fips)

bound_blocks = gpd.read_file(
    block_boundary_path, sql=f"SELECT * FROM block_boundary  WHERE COUNTYFP20 IN({county_list_sql})"

Show code 
district_polys = bound_districts[["name", "geometry"]]
county_polys = bound_counties[["COUNTYFP", "geometry"]]

# Point sources join directly to polygons, so no blocks are involved
point_sources = [
    ("school", "UGRC", ugrc_schools, "schoolid"),
    ("school", "NCES public", nces_public_schools, "NCESSCH"),
    ("school", "NCES private", nces_private_schools, "PPIN"),
    ("college", "UGRC", ugrc_colleges, "xid"),
    ("college", "NCES", nces_colleges, "UNITID"),
]

rows = []
for unit, source, pts, id_col in point_sources:
    point_district = gpd.sjoin(pts[[id_col, "geometry"]], district_polys, predicate="within", how="left")
    point_county = gpd.sjoin(pts[[id_col, "geometry"]], county_polys, predicate="within", how="left")
    assert len(point_district) == len(pts) == len(point_county), f"{source} {unit}: duplicate matches"
    rows.append(pd.DataFrame({
        "Unit": unit,
        "Source": source,
        "ID": pts[id_col].astype(str).values,
        "District Name": point_district["name"].values,
        "COUNTYFP": point_county["COUNTYFP"].values,
    }))

# MAZ rows: district and county from the MAZ centroid, so MAZs with no blocks are covered
maz_points = bound_maz.set_geometry(bound_maz.centroid)[["maz_id", "geometry"]]
maz_district = gpd.sjoin(maz_points, district_polys, predicate="within", how="left")
maz_county = gpd.sjoin(maz_points, county_polys, predicate="within", how="left")
assert len(maz_district) == len(maz_points) == len(maz_county), "MAZ centroid matches more than one polygon"
rows.append(pd.DataFrame({
    "Unit": "maz",
    "Source": "PopulationSim",
    "ID": maz_points["maz_id"].astype(str).values,
    "District Name": maz_district["name"].values,
    "COUNTYFP": maz_county["COUNTYFP"].values,
}))

long = pd.concat(rows, ignore_index=True)
school_crosswalk = long[long["Unit"].isin(["school", "maz"])].copy()
college_crosswalk = long[long["Unit"].isin(["college", "maz"])].copy()

for name, df in [("school", school_crosswalk), ("college", college_crosswalk)]:
    print(name, "rows:", len(df), "duplicate (Source, ID):", df.duplicated(["Source", "ID"]).sum())

school_crosswalk.to_csv(paths["intermediate"] / "crosswalk_school.csv", index=False)
college_crosswalk.to_csv(paths["intermediate"] / "crosswalk_college.csv", index=False)

10. Check the crosswalk

Every later join depends on these match rates, so coverage is checked here before anything downstream uses the crosswalk.

The map is drawn in OJS, and the coverage summary is printed by Python. Python first writes the map layers as GeoJSON (WGS84, which is what browsers expect) so OJS can read them.
