Geography

School district, county, and college-location crosswalks used by every later step

Overview

This script builds the geographic crosswalk that every later step in this project joins against.

  • Objective: tie every Census block in the Wasatch Front region (Box Elder, Davis, Weber, Salt Lake, and Utah counties) to the school district and county it falls in, and to the K-12 school(s) and college(s) located in it — so that PopulationSim’s synthetic population (3_synthetic.qmd) and real enrollment data (2_enrollment.qmd) can both be rolled up to district/county level as exact joins, with no area-weighting or estimation involved.
  • What it pulls: county, school district, Travel Analysis Zone (TAZ), and Census block boundaries; K-12 school and college locations from both UGRC/SGID (the state-preferred source) and NCES (kept alongside, not as a replacement, so schools and colleges can also be joined back to federal CCD/IPEDS data by ID); and the WFRC/MAG travel demand model boundary, used later to handle Box Elder County’s partial coverage in that model. TAZ boundaries come from WFRC’s WasatchFrontTAZ feature service rather than the TDM-GEO-AnalysisZones repo directly, so the pull stays current without tracking that repo’s releases — TAZID is what resolves 3_synthetic.qmd’s open question of how to join PopulationSim’s MAZ/TAZ-keyed output to this block-level crosswalk.
  • Output: a single block-level crosswalk (_data/intermediate/crosswalk.csv) carrying each block’s county, district, TAZ, and matched school/college IDs from every source.

Setup

import sqlalchemy
import geopandas as gpd
from pygris import blocks, counties
from pygris.helpers import validate_state, validate_county
from pathlib import Path
paths = {
    "raw": Path("_data/raw/"),
    "intermediate": Path("_data/intermediate/"),
    "output": Path("_outputs/"),
}

constants = {
    "state": "UT",
    "counties": {"Box Elder", "Davis", "Weber", "Salt Lake", "Utah"},
    "crs": "EPSG:26912",
}
opensgid_engine = sqlalchemy.create_engine(
    "postgresql+psycopg://agrc:agrc@opensgid.ugrc.utah.gov:5432/opensgid"
)
# Derive FIPS codes via pygris instead of hardcoding them
state_fips = validate_state(constants["state"])
county_fips = [
    validate_county(state_fips, county)
    for county in constants["counties"]
]
fips_codes = [f"{state_fips}{county}" for county in county_fips]

1. County boundaries

county_boundary_path = paths["raw"] / "geometry" / "county_boundary.gpkg"

# 1. Download only if not already cached locally
if not county_boundary_path.exists():
    all_counties = counties(state=state_fips, year=2023, cache=True).to_crs(constants["crs"])
    all_counties.to_file(county_boundary_path)

# 2. Always read from local storage
county_list_sql = ", ".join(f"'{fips}'" for fips in county_fips)

bound_counties = gpd.read_file(
    county_boundary_path, sql=f"SELECT * FROM county_boundary WHERE COUNTYFP IN ({county_list_sql})"
)
bound_counties.explore()

2. School district boundaries

district_boundary_path = paths["raw"] / "geometry" / "school_district_boundary.gpkg"

# 1. Download only if not already cached locally
if not district_boundary_path.exists():
    all_districts = gpd.read_postgis(
        "select * from boundaries.school_district_boundaries", opensgid_engine, geom_col="shape"
    ).to_crs(constants["crs"])

    all_districts["county_fips"] = gpd.sjoin(
        all_districts.set_geometry(all_districts.centroid),
        all_counties[["COUNTYFP", "geometry"]],
        predicate="within",
        how="left",
    )["COUNTYFP"]

    all_districts.to_file(district_boundary_path)

# 2. Read + filter in one step, pushed down to SQLite
county_list_sql = ", ".join(f"'{fips}'" for fips in county_fips)

bound_districts = gpd.read_file(
    district_boundary_path,
    sql=f"SELECT * FROM school_district_boundary WHERE county_fips IN ({county_list_sql})",
)
bound_districts.explore()

5. Census blocks

block_boundary_path = paths["raw"] / "geometry" / "block_boundary.gpkg"

# 1. Download only if not already cached locally
if not block_boundary_path.exists():
    all_blocks = blocks(state=state_fips, county=county_fips, year=2023, cache=True).to_crs(
        constants["crs"]
    )

    all_blocks["district_name"] = gpd.sjoin(
        all_blocks.set_geometry(all_blocks.centroid),
        all_districts[["name", "shape"]],
        predicate="within",
        how="left",
    )["name"]

    all_blocks["TAZID"] = gpd.sjoin(
        all_blocks.set_geometry(all_blocks.centroid),
        all_taz[["TAZID", "geometry"]],
        predicate="within",
        how="left",
    )["TAZID"]

    all_blocks.to_file(block_boundary_path)

# 2. Read + filter in one step, pushed down to SQLite
county_list_sql = ", ".join(f"'{fips}'" for fips in county_fips)

bound_blocks = gpd.read_file(
    block_boundary_path, sql=f"SELECT * FROM block_boundary WHERE COUNTYFP20 IN ({county_list_sql})"
)
bound_blocks.explore()

9. Generate crosswalks

school_block_match = gpd.sjoin(
    ugrc_schools[["schoolid", "geometry"]],
    bound_blocks[["GEOID20", "geometry"]],
    predicate="within",
    how="left",
)[["schoolid", "GEOID20"]]

college_block_match = gpd.sjoin(
    ugrc_colleges[["xid", "geometry"]],
    bound_blocks[["GEOID20", "geometry"]],
    predicate="within",
    how="left",
)[["xid", "GEOID20"]].rename(columns={"xid": "College ID"})

nces_public_block_match = gpd.sjoin(
    nces_public_schools[["NCESSCH", "geometry"]],
    bound_blocks[["GEOID20", "geometry"]],
    predicate="within",
    how="left",
)[["NCESSCH", "GEOID20"]].rename(columns={"NCESSCH": "NCES School ID"})

nces_private_block_match = gpd.sjoin(
    nces_private_schools[["PPIN", "geometry"]],
    bound_blocks[["GEOID20", "geometry"]],
    predicate="within",
    how="left",
)[["PPIN", "GEOID20"]].rename(columns={"PPIN": "NCES Private School ID"})

nces_college_block_match = gpd.sjoin(
    nces_colleges[["UNITID", "geometry"]],
    bound_blocks[["GEOID20", "geometry"]],
    predicate="within",
    how="left",
)[["UNITID", "GEOID20"]].rename(columns={"UNITID": "NCES College ID"})

crosswalk = (
    bound_blocks[["GEOID20", "COUNTYFP20", "district_name", "TAZID"]]
    .merge(school_block_match.rename(columns={"schoolid": "School ID"}), on="GEOID20", how="left")
    .merge(college_block_match, on="GEOID20", how="left")
    .merge(nces_public_block_match, on="GEOID20", how="left")
    .merge(nces_private_block_match, on="GEOID20", how="left")
    .merge(nces_college_block_match, on="GEOID20", how="left")
    .rename(
        columns={"GEOID20": "GEOID", "COUNTYFP20": "COUNTYFP", "district_name": "District Name"}
    )
)

crosswalk["School ID"] = crosswalk["School ID"].astype("Int64")
crosswalk["College ID"] = crosswalk["College ID"].astype("Int64")
crosswalk["TAZID"] = crosswalk["TAZID"].astype("Int64")

crosswalk.to_csv(paths["intermediate"] / "crosswalk.csv", index=False)
