Enrollment

Real K-12 (Utah DOE + NCES) and college (USHE + IPEDS) enrollment, by district and county

Overview

This script produces the “real enrollment” half of the validation schema — the numbers PopulationSim’s synthetic population (3_synthetic.qmd) gets compared against.

  • Objective: get district-level K-12 enrollment and county-level college enrollment from real, non-synthetic sources, joined through the school and college crosswalks built in 1_geography.qmd.
  • What it pulls: Utah DOE enrollment (public schools only, from UGRC’s own per-school counts), NCES enrollment by district (public via CCD, private via the Private School Universe Survey); USHE (student home ZIP code, not institution location — a different geography than everything else in this project) and NCES/IPEDS enrollment by county.
  • Known gaps: NCES private-school enrollment is school-location-based, not residence-based, and a handful of statewide online-only private programs still need manual review before being trusted at face value (see section 3); USHE data is Academic Year 2025, not 2023-24 — no 2023-24 USHE extract exists in what WFRC has on file, so this is a real year mismatch against the ~2023 vintage used everywhere else, not an oversight.
  • Output: one table of district-level enrollment, one table of county-level enrollment, both written to _outputs/.

1. K-12 — Utah DOE enrollment by district

Show code 
# Public schools only
ugrc_public_schools = gpd.read_file(
    paths["raw"] / "geometry" / "ugrc_school.gpkg",
    sql="""
        SELECT * FROM ugrc_school
        WHERE privateschool = 'False'
          AND onlineschool IS NOT 'Y'
    """,
)

# Join school -> district via the school crosswalk (point-in-polygon, one row per school) rather than
# matching the school's own `leaname` text against the district boundary's `name` text.
doe_crosswalk = school_crosswalk[
    (school_crosswalk["Unit"] == "school") & (school_crosswalk["Source"] == "UGRC")
]
assert (doe_crosswalk.groupby("ID")["District Name"].nunique() <= 1).all(), (
    "UGRC school ID maps to two districts"
)
doe_crosswalk = doe_crosswalk.drop_duplicates(subset=["ID"])

# The UGRC layer repeats some school IDs. Exact copies (Minersville, 201, is listed twice with the same
# count) are dropped so they are not double-counted. Separate records that share an ID (Grouse Creek,
# 218, an elementary and a secondary with different counts) are both kept and summed.
doe_grade_cols = ["preschool", "k"] + [f"grade{i}" for i in range(1, 13)]
ugrc_unique = (
    ugrc_public_schools[["schoolid", "schoolname", "totalk12", *doe_grade_cols]]
    .drop_duplicates()
    .assign(doe_students=lambda df: df["totalk12"] + df["preschool"].fillna(0))
)

# Left join, so schools with no crosswalk match are kept visible instead of silently dropped
doe_schools = (
    ugrc_unique.assign(schoolid=lambda df: df["schoolid"].astype("Int64").astype(str))
    .merge(doe_crosswalk[["ID", "District Name"]], how="left", left_on="schoolid", right_on="ID")
)
assert len(doe_schools) == len(ugrc_unique), "school join changed the row count"

unmatched_doe_schools = doe_schools[doe_schools["District Name"].isna()]
print(
    f"DOE schools with no district match: {len(unmatched_doe_schools)} "
    f"({unmatched_doe_schools['doe_students'].sum():,.0f} students, excluded from district totals)"
)

# groupby drops the unmatched rows (NaN district), so they are excluded from the totals below
doe_enrollment_by_district = (
    doe_schools.groupby("District Name")["doe_students"]
    .sum()
    .rename("doe_enrollment")
    .reset_index()
)

2. K-12 — NCES enrollment by district

Show code 
# Matches constants["data_year"] -- the vintage pulled in 1_geography.qmd
nces_public_school = gpd.read_file(paths["raw"] / "geometry" / "nces_public_school.gpkg")

nces_crosswalk = school_crosswalk[school_crosswalk["Source"] == "NCES public"]
assert (nces_crosswalk.groupby("ID")["District Name"].nunique() <= 1).all(), (
    "NCES public ID maps to two districts"
)
nces_crosswalk = nces_crosswalk.drop_duplicates(subset=["ID"])

# DOE totalk12 is K-12 only: it equals the sum of k and grades 1-12 on every public, non-online row, and
# it excludes preschool. NCES TOTAL includes PreK, plus other codes (UG, AE, G13), so NCES is summed
# over KG-G12 to match the DOE definition. NCES codes a grade the school doesn't offer as -2; those
# cells count as zero students.
nces_k12_grades = ["KG"] + [f"G{i:02d}" for i in range(1, 13)]
assert (nces_public_school[nces_k12_grades] >= -2).all().all(), "NCES K-12 grade has an unexpected code"
nces_public_school["K12"] = nces_public_school[nces_k12_grades].clip(lower=0).fillna(0).sum(axis=1)

# PK (preschool) uses the same -2 "not offered" rule as the grades, so negative codes count as zero
assert (nces_public_school["PK"] >= -2).all(), "NCES PK has an unexpected code"
nces_public_school["PK_students"] = nces_public_school["PK"].clip(lower=0).fillna(0)
nces_public_school["nces_students"] = nces_public_school["K12"] + nces_public_school["PK_students"]

nces_enrollment_by_district = (
    nces_public_school[["NCESSCH", "nces_students"]]
    .drop_duplicates()
    .assign(NCESSCH=lambda df: df["NCESSCH"].astype(str))
    .merge(
        nces_crosswalk[["ID", "District Name"]],
        left_on="NCESSCH",
        right_on="ID",
    )
    .groupby("District Name")["nces_students"]
    .sum()
    .rename("nces_enrollment")
    .reset_index()
)

4. Colleges — USHE enrollment by county

Show code 
# NOTE: the only USHE data WFRC has on file is Academic Year 2025 (Term 2 / Spring) and Academic Year 2020 (Term 2)
# USHE covers Utah's PUBLIC institutions only. Private institutions (e.g., BYU, Westminster, Ensign)
# are not in USHE and only appear in the IPEDS county totals, so county comparisons are not like-for-like.
# Both full-time and part-time students are kept, matching the WF-TDM college repo (6-CollegeControlTotals.qmd).
ushe = pd.read_excel(paths["raw"] / "ushe" / "WFRC Data Request_Final.xlsx")

ushe_2025_fall = ushe.query("`Academic Year` == 2025 and Term == 2")
Show code

    ("Utah State University", "USU - Brigham Center"): 14,
    ("Utah State University", "USU - Ogden Center"): 1,
    ("Utah State University", "USU - Orem Center"): 28,
    ("Utah State University", "USU - Salt Lake Center"): 32,
    ("Utah State University", "USU - Tremonton Center"): 31,
    ("Utah Valley University", "Main Campus"): 5,

    ("Weber State University", "Technical College Campus--Davis"): 56,
    ("Weber State University", "Technical College Campus--Ogden-Weber"): 69,
    ("Weber State University", "University Center--SLCC"): 4,

5. Colleges — NCES/IPEDS enrollment by county

Show code 
# IPEDS Fall Enrollment distance-education component (EF{year}A_DIST). In-person enrollment is
# students in no distance-ed courses (EFDENON) plus students in some distance-ed / hybrid courses
# (EFDESOM); exclusively online students are excluded because they generate no campus trips.
# Same source and method as the WF-TDM college enrollment repo (5-CollegeEnrollmentIPEDS.qmd).
ipeds_dir = paths["raw"] / "ipeds"
ipeds_dir.mkdir(parents=True, exist_ok=True)
zip_path = ipeds_dir / f"EF{constants['data_year']}A_DIST.zip"

if not zip_path.exists():
    # The datacenter/data path serves 2023; the tablefiles/zipfiles path used in the WF-TDM repo returns 404
    response = requests.get(
        f"https://nces.ed.gov/ipeds/datacenter/data/EF{constants['data_year']}A_DIST.zip"
    )
    response.raise_for_status()
    zip_path.write_bytes(response.content)

# pandas reads the single CSV inside the zip directly, so nothing is extracted to disk
ipeds_dist = pd.read_csv(zip_path)
Show code 
# EFDELEV == 1 is the grand-total row for each institution (the other levels are breakdowns)
ipeds_in_person = ipeds_dist.query("EFDELEV == 1").assign(
    in_person=lambda df: df["EFDENON"].fillna(0) + df["EFDESOM"].fillna(0)
)[["UNITID", "in_person"]]

# The seven target colleges from the WF-TDM college repo (6-CollegeControlTotals.qmd), each assigned
# to its county. The other institutions in the crosswalk are left out to match that snapshot.
target_county = {
    230764: "035",  # University of Utah (Salt Lake)
    230746: "035",  # Salt Lake Community College (all five campuses are in Salt Lake County)
    230418: "035",  # Ensign College
    230807: "035",  # Westminster University
    230038: "049",  # Brigham Young University (Utah County)
    230737: "049",  # Utah Valley University (Utah County)
    230782: "057",  # Weber State University (Main campus, Weber County)
}

ipeds_targets = ipeds_in_person[ipeds_in_person["UNITID"].isin(target_county)].assign(
    COUNTYFP=lambda df: df["UNITID"].map(target_county)
)

# Weber State's Davis campus is counted in Davis County, using its share of USHE headcount among
# the two WSU campuses in the repo's split (Main campus and WSU - Davis Center)
wsu_sites = ushe_2025_fall[
    (ushe_2025_fall["Institution"] == "Weber State University")
    & ushe_2025_fall["Site"].isin(["Main Campus", "WSU - Davis Center"])
].groupby("Site")["Headcount"].sum()
wsu_davis_share = wsu_sites["WSU - Davis Center"] / wsu_sites.sum()

wsu_total = ipeds_targets.loc[ipeds_targets["UNITID"] == 230782, "in_person"].item()
wsu_davis_students = round(wsu_total * wsu_davis_share)
ipeds_targets.loc[ipeds_targets["UNITID"] == 230782, "in_person"] = wsu_total - wsu_davis_students
ipeds_targets = pd.concat(
    [
        ipeds_targets,
        pd.DataFrame({"UNITID": [230782], "in_person": [wsu_davis_students], "COUNTYFP": ["011"]}),
    ],
    ignore_index=True,
)

ipeds_enrollment_by_county = (
    ipeds_targets.groupby("COUNTYFP")["in_person"]
    .sum()
    .rename("nces_enrollment")
    .reset_index()
)

# Canonical district list -- every district the school crosswalk covers (school points and MAZ rows),
# regardless of whether a given source has a school in it (left-merging onto this keeps a real data
# gap visible as NaN instead of silently dropping the row via an inner join)
district_list = school_crosswalk[["District Name"]].dropna().drop_duplicates()

school_enrollment = (
    district_list.merge(doe_enrollment_by_district, on="District Name", how="left")
    .merge(nces_enrollment_by_district, on="District Name", how="left")
    .merge(nces_private_enrollment_by_district, on="District Name", how="left")
)
# No private schools in a district is a real zero, not a gap
school_enrollment["nces_private_enrollment"] = school_enrollment["nces_private_enrollment"].fillna(
    0
)

School Enrollment by Level

Show code 
# Enrollment by school level, summed from the grade columns each level covers. A school's own label
# can't split its students by level, since one school can serve several levels.
# Each level maps to its DOE columns and its NCES columns.
school_levels = {
    "Pre-K": (["preschool"], ["PK"]),
    "K": (["k"], ["KG"]),
    "Elementary": ([f"grade{i}" for i in range(1, 6)], [f"G{i:02d}" for i in range(1, 6)]),
    "Middle": ([f"grade{i}" for i in range(6, 9)], [f"G{i:02d}" for i in range(6, 9)]),
    "High": ([f"grade{i}" for i in range(9, 13)], [f"G{i:02d}" for i in range(9, 13)]),
}

doe_by_level = (
    doe_schools.assign(
        **{level: doe_schools[doe_cols].fillna(0).sum(axis=1) for level, (doe_cols, _) in school_levels.items()}
    )
    .groupby("District Name")[list(school_levels)]
    .sum()
)

# NCES negative codes (-2, grade not offered) count as zero, same as the K-12 totals above
nces_levels = nces_public_school[["NCESSCH"]].assign(
    **{
        level: nces_public_school[nces_cols].clip(lower=0).fillna(0).sum(axis=1)
        for level, (_, nces_cols) in school_levels.items()
    }
)
nces_by_level = (
    nces_levels.drop_duplicates()
    .assign(NCESSCH=lambda df: df["NCESSCH"].astype(str))
    .merge(nces_crosswalk[["ID", "District Name"]], left_on="NCESSCH", right_on="ID")
    .groupby("District Name")[list(school_levels)]
    .sum()
)

real_by_level = (
    doe_by_level.reset_index()
    .melt(id_vars="District Name", var_name="Level", value_name="doe_enrollment")
    .merge(
        nces_by_level.reset_index().melt(
            id_vars="District Name", var_name="Level", value_name="nces_enrollment"
        ),
        on=["District Name", "Level"],
        how="outer",
    )
)

# The levels must add back up to the district totals computed above
doe_gap = (
    real_by_level.groupby("District Name")["doe_enrollment"].sum()
    - doe_enrollment_by_district.set_index("District Name")["doe_enrollment"]
)
nces_gap = (
    real_by_level.groupby("District Name")["nces_enrollment"].sum()
    - nces_enrollment_by_district.set_index("District Name")["nces_enrollment"]
)
assert doe_gap.abs().max() < 0.5, "DOE levels do not add up to district totals"
assert nces_gap.abs().max() < 0.5, "NCES levels do not add up to district totals"

real_by_level.to_csv(paths["intermediate"] / "real_by_district_level.csv", index=False)

}

# IPEDS, fall 2023 (same definitions as section 5)
ipeds_long = (
    ipeds_dist[ipeds_dist["EFDELEV"] == 1]
    .loc[lambda df: df["UNITID"].isin(list(university_names))]
    .assign(
        Institution=lambda df: df["UNITID"].map(university_names),
        **{
            "In-person only": lambda df: df["EFDENON"].fillna(0),
            "Hybrid (some DE)": lambda df: df["EFDESOM"].fillna(0),
            "Online only": lambda df: df["EFDEEXC"].fillna(0),
        },
    )
    .melt(
        id_vars=["Institution"],
        value_vars=["In-person only", "Hybrid (some DE)", "Online only"],
        var_name="Segment",
        value_name="Students",
    )
    .assign(Source="IPEDS")
)

# USHE, Academic Year 2025 Term 2 (full- and part-time). The "taught online" rows are the hybrid
# portion; USHE has no online-only category.
ushe_long = (
    ushe_2025_fall[ushe_2025_fall["Institution"].isin(list(ushe_names))]
    .assign(
        Institution=lambda df: df["Institution"].map(ushe_names),
        Segment=lambda df: df["Site"].eq("This portion of the course is taught online").map(
            {True: "Hybrid (some DE)", False: "In-person only"}
        ),
    )
    .groupby(["Institution", "Segment"], as_index=False)["Headcount"]
    .sum()
    .rename(columns={"Headcount": "Students"})
    .assign(Source="USHE")
)

modality_compare = pd.concat([ipeds_long, ushe_long], ignore_index=True)[
    ["Institution", "Source", "Segment", "Students"]
