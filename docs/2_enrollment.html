Enrollment

Real K-12 (Utah DOE + NCES) and college (USHE + IPEDS) enrollment, by district and county

Overview

This script produces the “real enrollment” half of the validation schema — the numbers PopulationSim’s synthetic population (3_synthetic.qmd) gets compared against.

  • Objective: get district-level K-12 enrollment and county-level college enrollment from real, non-synthetic sources, joined through the block-level crosswalk built in 1_geography.qmd.
  • What it pulls: Utah DOE enrollment (public schools only, from UGRC’s own per-school counts), NCES enrollment by district (public via CCD, private via the Private School Universe Survey); USHE (student home ZIP code, not institution location — a different geography than everything else in this project) and NCES/IPEDS enrollment by county.
  • Known gaps: NCES private-school enrollment is school-location-based, not residence-based, and a handful of statewide online-only private programs still need manual review before being trusted at face value (see section 3); USHE data is Academic Year 2025, not 2023-24 — no 2023-24 USHE extract exists in what WFRC has on file, so this is a real year mismatch against the ~2023 vintage used everywhere else, not an oversight.
  • Output: one table of district-level enrollment, one table of county-level enrollment, both written to _outputs/.

Setup

import pandas as pd
import geopandas as gpd
import requests
import zipfile
from pathlib import Path

paths = {
    "raw": Path("_data/raw/"),
    "intermediate": Path("_data/intermediate/"),
    "output": Path("_outputs/"),
}

constants = {
    "data_year": 2023  # NCES/IPEDS/PSS vintage -- bump this one number in ~4 years
}
year_code = f"{str(constants['data_year'])[-2:]}{str(constants['data_year'] + 1)[-2:]}"

crosswalk = pd.read_csv(paths["intermediate"] / "crosswalk.csv", dtype={"COUNTYFP": str})

1. K-12 — Utah DOE enrollment by district

# Public schools only
ugrc_public_schools = gpd.read_file(
    paths["raw"] / "geometry" / "ugrc_school.gpkg",
    sql="""
        SELECT * FROM ugrc_school
        WHERE privateschool = 'False'
          AND onlineschool IS NOT 'Y'
    """,
)

# Join school -> district via the crosswalk (block-based, spatial) rather than matching school's own `leaname` text against the district boundary's `name` text
doe_enrollment_by_district = (
    ugrc_public_schools[["schoolid", "totalk12"]]
    .merge(crosswalk[["School ID", "District Name"]], left_on="schoolid", right_on="School ID")
    .groupby("District Name")["totalk12"]
    .sum()
    .rename("doe_enrollment")
    .reset_index()
)

2. K-12 — NCES enrollment by district

# Matches constants["data_year"] -- the vintage pulled in 1_geography.qmd
nces_public_school = gpd.read_file(paths["raw"] / "geometry" / "nces_public_school.gpkg")

# Dedup crosswalk for multiple NCES schools at same location
crosswalk_nces = (
    crosswalk.dropna(subset=["NCES School ID"])
    .drop_duplicates(subset=["NCES School ID"])
    .assign(**{"NCES School ID": lambda df: df["NCES School ID"].astype("Int64").astype(str)})
)

nces_enrollment_by_district = (
    nces_public_school[["NCESSCH", "TOTAL"]]
    .merge(
        crosswalk_nces[["NCES School ID", "District Name"]],
        left_on="NCESSCH",
        right_on="NCES School ID",
    )
    .groupby("District Name")["TOTAL"]
    .sum()
    .rename("nces_enrollment")
    .reset_index()
)

3. K-12 — NCES private-school enrollment by district

# Private School Universe Survey (PSS)
pss_dir = paths["raw"] / "pss"
pss_dir.mkdir(parents=True, exist_ok=True)
pss_csv_path = pss_dir / f"pss{year_code}_pu.csv"

if not pss_csv_path.exists():
    zip_path = pss_dir / f"pss{year_code}_pu_csv.zip"
    zip_path.write_bytes(
        requests.get(f"https://nces.ed.gov/surveys/pss/zip/pss{year_code}_pu_csv.zip").content
    )
    with zipfile.ZipFile(zip_path) as z:
        z.extractall(pss_dir)

# NUMSTUDS = total enrollment (verified 2026-10-01 against NCES's own SIZE bucket breaks for every
# UT row checked). PPIN is a string -- some IDs are alpha-prefixed (e.g. "A2392167"), don't cast to
# numeric.
pss = pd.read_csv(
    pss_csv_path, usecols=["PPIN", "PSTABB", "PINST", "NUMSTUDS"], dtype={"PPIN": str}
).query("PSTABB == 'UT'")
# Same duplicate-schools issue as the public crosswalk merge -- dedup on NCES Private School ID first.
crosswalk_private = crosswalk.dropna(subset=["NCES Private School ID"]).drop_duplicates(
    subset=["NCES Private School ID"]
)

nces_private_enrollment_by_district = (
    pss[["PPIN", "PINST", "NUMSTUDS"]]
    .merge(
        crosswalk_private[["NCES Private School ID", "District Name"]],
        left_on="PPIN",
        right_on="NCES Private School ID",
    )
    .groupby("District Name")["NUMSTUDS"]
    .sum()
    .rename("nces_private_enrollment")
    .reset_index()
)

4. Colleges — USHE enrollment by county

# NOTE: the only USHE data WFRC has on file is Academic Year 2025 (Term 2 / Spring) and Academic Year 2020 (Term 2)
ushe = pd.read_excel(paths["raw"] / "ushe" / "WFRC Data Request_Final.xlsx")

ushe_fulltime = ushe.query(
    "`Academic Year` == 2025 and Term == 2 and Full_or_Part_Time == 'Full-Time'"
)
# Scoped to WFRC region institutions/sites only (constants["counties"] in 1_geography.qmd) --
# USHE has no surrogate ID, so (Institution, Site) is the natural key; UGRC's unique ID is `xid`.
# A few sites have no UGRC point of their own and are reassigned to a same-county campus instead --
# "USU - Ogden Center" goes to Weber State's Main Campus xid, NOT USU's own (Cache County) Main
# Campus, since the reassignment target has to match the real site's county, not its institution.
ushe_to_ugrc_xid = {
    ("Salt Lake Community College", "Correctional Facility--West Valley City"): 52,
    ("Salt Lake Community College", "Main Campus"): 4,
    ("Salt Lake Community College", "SLCC - Airport Center"): 4,
    ("Salt Lake Community College", "SLCC - Jordan Campus"): 51,
    ("Salt Lake Community College", "SLCC - Larry H. Miller Campus"): 50,
    ("Salt Lake Community College", "SLCC - South City Center"): 46,
    ("Salt Lake Community College", "SLCC - West Point Center"): 78,
    ("Salt Lake Community College", "SLCC - West Valley City"): 52,
    ("Salt Lake Community College", "SLCC-UU Juniper building at Herriman Campus"): 4,
    ("University Of Utah", "Main Campus"): 7,
    ("University Of Utah", "SLCC-UU Juniper building at Herriman Campus"): 7,
    ("University Of Utah", "UU - Sandy Center"): 7,
    ("Utah State University", "USU - Brigham Center"): 14,
    ("Utah State University", "USU - Ogden Center"): 1,
    ("Utah State University", "USU - Orem Center"): 28,
    ("Utah State University", "USU - Salt Lake Center"): 32,
    ("Utah State University", "USU - Tremonton Center"): 31,
    ("Utah Valley University", "Main Campus"): 5,
    ("Weber State University", "Main Campus"): 1,
    ("Weber State University", "Technical College Campus--Davis"): 56,
    ("Weber State University", "Technical College Campus--Ogden-Weber"): 69,
    ("Weber State University", "University Center--SLCC"): 4,
    ("Weber State University", "WSU - Davis Center"): 37,
    ("Weber State University", "WSU - Farmington Station"): 42,
}

# Everything not in the dict is out-of-region or a non-geographic site (online, other-site, etc.)
ushe_in_region = ushe_fulltime[
    ushe_fulltime.set_index(["Institution", "Site"]).index.isin(ushe_to_ugrc_xid)
].assign(xid=lambda df: df.set_index(["Institution", "Site"]).index.map(ushe_to_ugrc_xid))

ugrc_colleges = gpd.read_file(paths["raw"] / "geometry" / "ugrc_college.gpkg")

ushe_enrollment_by_county = (
    ushe_in_region.merge(ugrc_colleges[["xid", "county_fips"]], on="xid")
    .groupby("county_fips")["Headcount"]
    .sum()
    .rename("ushe_enrollment")
    .reset_index()
)

5. Colleges — NCES/IPEDS enrollment by county

# IPEDS Fall Enrollment survey component (EF) -- matches constants["data_year"] across the project.
ipeds_dir = paths["raw"] / "ipeds"
ipeds_dir.mkdir(parents=True, exist_ok=True)
ipeds_csv_path = ipeds_dir / f"ef{constants['data_year']}a.csv"

if not ipeds_csv_path.exists():
    zip_path = ipeds_dir / f"EF{constants['data_year']}A.zip"
    zip_path.write_bytes(
        requests.get(
            f"https://nces.ed.gov/ipeds/datacenter/data/EF{constants['data_year']}A.zip"
        ).content
    )
    with zipfile.ZipFile(zip_path) as z:
        z.extractall(ipeds_dir)

ipeds_ef = pd.read_csv(ipeds_csv_path)
# EFALEVEL == 1 is the grand-total-headcount row (verified against a known institution's total
# before relying on it); every other EFALEVEL value is a breakdown (undergrad/grad/level/etc.)
ipeds_totals = ipeds_ef.query("EFALEVEL == 1")[["UNITID", "EFTOTLT"]]

nces_college = gpd.read_file(paths["raw"] / "geometry" / "nces_college.gpkg")

ipeds_enrollment_by_county = (
    ipeds_totals.merge(
        crosswalk[["NCES College ID", "COUNTYFP"]].drop_duplicates(),
        left_on="UNITID",
        right_on="NCES College ID",
    )
    .groupby("COUNTYFP")["EFTOTLT"]
    .sum()
    .rename("ipeds_enrollment")
    .reset_index()
)

6. Assemble enrollment tables

School Enrollment by School Districts

# Canonical district list -- every district the block-level crosswalk covers, regardless of
# whether a given source has a school in it (left-merging onto this keeps a real data gap visible
# as NaN instead of silently dropping the row via an inner join)
district_list = crosswalk[["District Name"]].dropna().drop_duplicates()

school_enrollment = (
    district_list.merge(doe_enrollment_by_district, on="District Name", how="left")
    .merge(nces_enrollment_by_district, on="District Name", how="left")
    .merge(nces_private_enrollment_by_district, on="District Name", how="left")
)
# No private schools in a district is a real zero, not a gap
school_enrollment["nces_private_enrollment"] = school_enrollment["nces_private_enrollment"].fillna(
    0
)
