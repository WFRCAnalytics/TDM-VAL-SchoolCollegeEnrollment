Validation

Real enrollment vs. PopulationSim synthetic estimates, by district and county

Overview

This is the page the site is built around: where the real enrollment numbers (2_enrollment.qmd) and PopulationSim’s synthetic estimates (3_synthetic.qmd) get compared.

  • Objective: for each school district, compare the synthetic enrolled count (school grades) and the synthetic population aged 3-18 against real enrollment from Utah DOE or NCES CCD. For each county, compare the same kind of synthetic counts against USHE or IPEDS. The population counts are a sanity bound, since age is controlled in PopulationSim, so they are not expected to match.
  • Using the page: the inputs above each table pick what is compared and against which source. The table and the scatter plots update with them. The flag marks rows that are further off than the threshold you set.
  • What it reads: the district and county tables from 2_enrollment.qmd and the synthetic tables from 3_synthetic.qmd, all in _data/intermediate/.
  • Status: sections 1 and 2 (districts and counties) are built. Section 3 (Box Elder) and the summary in section 4 are still placeholders.
  • Box Elder: reported separately, since its real and synthetic numbers are both scoped differently there (model region vs. full county). It is still a TODO.

// Tooltip colours follow the page theme. Plot draws each tooltip box with a fixed background and its
// text inherits the page colour, so the box colour has to match the theme. This re-runs when the theme
// changes (the observers below), so the charts redraw with the new tooltip colours.
tipTheme = Generators.observe(notify => {
  const backgroundOf = el => {
    const bg = getComputedStyle(el).backgroundColor;
    return bg.startsWith("rgba(0, 0, 0, 0)") || bg === "transparent" ? null : bg;
  };
  const isDark = () => {
    const bg = backgroundOf(document.body) ?? backgroundOf(document.documentElement) ?? "rgb(255, 255, 255)";
    const [r, g, b] = bg.match(/[\d.]+/g).slice(0, 3).map(Number);
    return 0.299 * r + 0.587 * g + 0.114 * b < 128;
  };
  const emit = () => notify(isDark()
    ? {fill: "#0c2a3b", stroke: "#4b5563"}
    : {fill: "#ffffff", stroke: "#374151"});
  const observer = new MutationObserver(emit);
  observer.observe(document.documentElement, {attributes: true, attributeFilter: ["class", "style", "data-bs-theme", "data-theme"]});
  observer.observe(document.body, {attributes: true, attributeFilter: ["class", "style"]});
  observer.observe(document.head, {subtree: true, attributes: true, attributeFilter: ["href", "disabled", "media"]});
  const media = window.matchMedia("(prefers-color-scheme: dark)");
  media.addEventListener("change", emit);
  emit();
  return () => {
    observer.disconnect();
    media.removeEventListener("change", emit);
  };
})

1. School validation (by district)

compareOptions = ["Enrolled (school grades)", "Population aged 3-18"];
referenceOptions = ["Utah DOE", "NCES CCD"];
measureOptions = ["Difference", "Percent difference", "Ratio"];

districtReal = FileAttachment("_data/intermediate/real_enrollment_by_district.csv").csv({typed: true});

districtSynthMap = new Map(districtSynth.map(d => [d["District Name"], d]));

districtRefKey = `Reference (${districtReference})`;
districtSynKey = districtCompare === compareOptions[0] ? "Synthetic enrolled" : "Synthetic population";

Show code 
districtTable = districtReal.map(d => {
  const s = districtSynthMap.get(d["District Name"]);
  const ref = districtReference === "Utah DOE" ? d.doe_enrollment : d.nces_enrollment;
  const syn = s ? (districtCompare === compareOptions[0] ? s.sch_grade : s.age_3_18) : null;
  const diff = syn != null && ref != null ? syn - ref : null;
  const pct = diff != null && ref ? diff / ref : null;
  const ratio = diff != null && ref ? syn / ref : null;
  const shown = districtMeasure === "Difference" ? diff : districtMeasure === "Percent difference" ? pct : ratio;
  return {
    District: d["District Name"],
    [districtRefKey]: ref,
    [districtSynKey]: syn,
    [districtMeasure]: shown,
    Flag: pct != null && Math.abs(pct) * 100 > districtFlag ? "⚠" : ""
  };
})

countFormat = v => v == null ? "—" : Math.round(v).toLocaleString("en-US");
shownFormat = districtMeasure === "Percent difference"
  ? (v => v == null ? "—" : `${(v * 100).toFixed(1)}%`)
  : districtMeasure === "Ratio"
    ? (v => v == null ? "—" : v.toFixed(2))
    : (v => v == null ? "—" : `${v > 0 ? "+" : ""}${Math.round(v).toLocaleString("en-US")}`);

Inputs.table(districtTable, {
  columns: ["District", districtRefKey, districtSynKey, districtMeasure, "Flag"],
  format: {[districtRefKey]: countFormat, [districtSynKey]: countFormat, [districtMeasure]: shownFormat},
  maxHeight: "none"
})

scatterRows = districtReal.flatMap(d => {

    {reference: "NCES CCD", measure: "Enrolled", x: d.nces_enrollment, y: s.sch_grade, district: name},
    {reference: "NCES CCD", measure: "Population", x: d.nces_enrollment, y: s.age_3_18, district: name}
  ];

    {reference, measure, x: 0, y: 0},
    {reference, measure, x: scatterMax, y: scatterMax}

Enrolled by school level

Pick any combination of school levels and districts. The chart adds up the selection: each district shows real DOE and NCES enrollment next to the synthetic grade count and the synthetic age-band count.

Show code 
levelOrder = ["Pre-K", "K", "Elementary", "Middle", "High"];

// Age bands used for the sanity bound: one band per level, no overlaps, so each age belongs to one level
ageBands = ({"Pre-K": [3, 4], "K": [5, 5], "Elementary": [6, 11], "Middle": [12, 14], "High": [15, 18]});

realByLevel = FileAttachment("_data/intermediate/real_by_district_level.csv").csv({typed: true});
synthByLevel = FileAttachment("_data/intermediate/synthetic_by_district_level.csv").csv({typed: true});
synthAge = FileAttachment("_data/intermediate/synthetic_age_by_district.csv").csv({typed: true});

levelRows = realByLevel.map(d => {
  const s = synthByLevel.find(x => x["District Name"] === d["District Name"] && x.Level === d.Level);
  return {
    district: d["District Name"],
    level: d.Level,
    doe: d.doe_enrollment ?? 0,
    nces: d.nces_enrollment ?? 0,
    grade: s ? s.sch_grade : 0
  };
});

districtNames = [...new Set(levelRows.map(d => d.district))].sort();
Show code 
viewof selectedLevels = Inputs.select(levelOrder, {
  label: "School levels",

  value: levelOrder,
  format: lvl => ageBands[lvl][0] === ageBands[lvl][1]
    ? `${lvl} (Age: ${ageBands[lvl][0]})`
    : `${lvl} (Age: ${ageBands[lvl][0]}-${ageBands[lvl][1]})`
})

viewof selectedDistricts = Inputs.select(districtNames, {
  label: "Districts",
  multiple: 5,
  value: districtNames
})

viewof splitByDistrict = Inputs.checkbox(["Split by district"], {
  label: "Chart",
  value: ["Split by district"]

selectedFacet = splitByDistrict.length > 0
Show code 
chartRows = {
  const out = [];
  for (const dist of selectedDistricts) {
    const rows = levelRows.filter(d => d.district === dist && selectedLevels.includes(d.level));
    if (!rows.length) continue;
    const ages = new Set();
    for (const lvl of selectedLevels) {
      const [lo, hi] = ageBands[lvl];
      for (let a = lo; a <= hi; a++) ages.add(a);
    }
    const ageTotal = d3.sum(
      synthAge.filter(d => d["District Name"] === dist && ages.has(d.age)),
      d => d.persons
    );
    out.push({district: dist, source: "DOE", students: d3.sum(rows, d => d.doe)});
    out.push({district: dist, source: "NCES", students: d3.sum(rows, d => d.nces)});
    out.push({district: dist, source: "Synthetic grade", students: d3.sum(rows, d => d.grade)});
    out.push({district: dist, source: "Synthetic age band", students: ageTotal});
  }
  if (selectedFacet) return out;
  // Split off: add the selected districts into one total per source
  return d3.rollups(out, v => d3.sum(v, d => d.students), d => d.source)
    .map(([source, students]) => ({district: "Total", source, students}));
}

chartRows.length === 0
  ? html`<p><em>Select at least one school level and one district.</em></p>`
  : Plot.plot({
      width: 960,
      height: 460,
      marginBottom: 60,
      grid: true,
      style: {background: "transparent", color: "inherit"},
      ...(selectedFacet ? {fx: {label: null, domain: selectedDistricts}} : {}),
      x: {axis: null},
      y: {label: "Students", grid: true},
      color: {
        legend: true,
        domain: ["DOE", "NCES", "Synthetic grade", "Synthetic age band"],
        range: ["#1f78b4", "#33a02c", "#e31a1c", "#ff7f00"]
      },
      marks: [
        Plot.barY(chartRows, {
          ...(selectedFacet ? {fx: "district"} : {}),
          x: "source",
          y: "students",
          fill: "source",
          tip: tipTheme
        }),
        Plot.ruleY([0])
      ]
    })

collegeTable = countyOrder.map(c => {
  const syn = synthCollege.find(d => d.COUNTYFP === c);

Colleges by county

Pick the counties and college levels to compare. USHE covers public institutions only and has no level split, so its bar always shows all levels. The age band counts persons aged 18-24 whatever the level, as a sanity bound.

  label: "Counties",
  multiple: 5,
  value: countyOrder,
  format: c => countyName[c]

  label: "College level",

Show code 
selectedCountySplit = splitByCounty.length > 0;
includePrivate = collegeIncludePrivate.length > 0;
ipedsLabel = includePrivate ? "IPEDS, all institutions" : "IPEDS, public only";
ipedsInScope = d => includePrivate || d.Control === "Public";
Show code 
collegeRows = {
  const out = [];
  for (const c of selectedCounties) {
    const ipeds = d3.sum(
      realCollegeLevel.filter(d => d.COUNTYFP === c && selectedCollegeLevels.includes(d.Level) && ipedsInScope(d)),
      d => d.nces_enrollment
    );
    const ushe = realCountyTotals.find(d => d.COUNTYFP === c)?.ushe_enrollment ?? 0;
    const syn = synthCollege.find(d => d.COUNTYFP === c) ?? {undergraduate: 0, graduate: 0, age_18_24: 0};
    const synEnrolled =
      (selectedCollegeLevels.includes("Undergraduate") ? syn.undergraduate : 0) +
      (selectedCollegeLevels.includes("Graduate") ? syn.graduate : 0);
    out.push({county: countyName[c], source: "USHE (all levels)", students: ushe});
    out.push({county: countyName[c], source: ipedsLabel, students: ipeds});
    out.push({county: countyName[c], source: "Synthetic enrolled", students: synEnrolled});
    out.push({county: countyName[c], source: "Synthetic age 18-24", students: syn.age_18_24});
  }
  if (selectedCountySplit) return out;
  // Split off: add the selected counties into one total per source
  return d3.rollups(out, v => d3.sum(v, d => d.students), d => d.source)
    .map(([source, students]) => ({county: "Total", source, students}));
}

      y: {label: "Students", grid: true},
      color: {
        legend: true,
